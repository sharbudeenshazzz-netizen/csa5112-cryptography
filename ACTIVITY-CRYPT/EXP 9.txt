#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main()
{
    char text[] = "KXJEYUREBEZWEHEWRYTUHEYFSKREHEGOYFIWTTTUOLKSYCAJPOBOTEIZONTXBYBNTGONEYCUZWRGDSONSXBOUYWRHEBAAHYUSEDQ";
    char key[] = "KENNEDY";
    char mat[5][5];
    int used[26] = {0};
    int i, j, k = 0, r1, c1, r2, c2;
    char ch;

    used['J' - 'A'] = 1;

    for (i = 0; key[i]; i++) {
        ch = toupper(key[i]);
        if (ch == 'J') ch = 'I';
        if (ch >= 'A' && ch <= 'Z' && !used[ch - 'A']) {
            mat[k / 5][k % 5] = ch;
            used[ch - 'A'] = 1;
            k++;
        }
    }

    for (ch = 'A'; ch <= 'Z'; ch++) {
        if (!used[ch - 'A']) {
            mat[k / 5][k % 5] = ch;
            k++;
        }
    }

    printf("Decrypted text: ");

    for (i = 0; text[i] && text[i + 1]; i += 2) {
        char a = text[i], b = text[i + 1];

        for (r1 = 0; r1 < 5; r1++)
            for (c1 = 0; c1 < 5; c1++)
                if (mat[r1][c1] == a) goto found1;

        found1:
        for (r2 = 0; r2 < 5; r2++)
            for (c2 = 0; c2 < 5; c2++)
                if (mat[r2][c2] == b) goto found2;

        found2:
        if (r1 == r2) {
            printf("%c%c", mat[r1][(c1 + 4) % 5],
                           mat[r2][(c2 + 4) % 5]);
        } else if (c1 == c2) {
            printf("%c%c", mat[(r1 + 4) % 5][c1],
                           mat[(r2 + 4) % 5][c2]);
        } else {
            printf("%c%c", mat[r1][c2], mat[r2][c1]);
        }
    }

    printf("\n");
    return 0;
}
