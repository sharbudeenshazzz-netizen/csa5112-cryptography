#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main() {
    char text[200], key[27];
    int i, seen[26] = {0};

    printf("Enter plaintext: ");
    fgets(text, sizeof(text), stdin);

    printf("Enter 26-letter substitution key: ");
    scanf("%26s", key);

    if (strlen(key) != 26) {
        printf("Key must contain 26 letters.\n");
        return 1;
    }

    for (i = 0; i < 26; i++) {
        key[i] = toupper((unsigned char)key[i]);

        if (key[i] < 'A' || key[i] > 'Z' ||
            seen[key[i] - 'A']) {
            printf("Invalid key: use each letter once.\n");
            return 1;
        }

        seen[key[i] - 'A'] = 1;
    }

    for (i = 0; text[i] != '\0'; i++) {
        if (isalpha((unsigned char)text[i])) {
            char base = isupper((unsigned char)text[i])
                        ? 'A' : 'a';
            text[i] = key[toupper((unsigned char)text[i]) - 'A'];

            if (base == 'a')
                text[i] = tolower((unsigned char)text[i]);
        }
    }

    printf("Ciphertext: %s", text);
    return 0;
}
