#include <stdio.h>
#include <ctype.h>
#include <string.h>

int main()
{
    char plain[100], cipher[100];
    char key[] = "CIPHERABDGJKLMNOQSTUVWXYZ";
    int i, x;

    printf("Enter plaintext: ");
    scanf("%99s", plain);

    for (i = 0; plain[i] != '\0'; i++)
    {
        x = tolower(plain[i]) - 'a';
        if (x >= 0 && x < 26)
            cipher[i] = key[x];
        else
            cipher[i] = plain[i];
    }

    cipher[i] = '\0';

    printf("Ciphertext: %s\n", cipher);

    return 0;
}
