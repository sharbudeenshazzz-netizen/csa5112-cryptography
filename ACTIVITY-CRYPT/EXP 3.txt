#include <stdio.h>
#include <string.h>
#include <ctype.h>

char mat[5][5];

void createMatrix(char key[]) {
    int used[26] = {0}, i, j, r = 0, c = 0;
    char ch;

    used['J' - 'A'] = 1;

    for (i = 0; key[i]; i++) {
        ch = toupper((unsigned char)key[i]);

        if (ch < 'A' || ch > 'Z') continue;
        if (ch == 'J') ch = 'I';

        if (!used[ch - 'A']) {
            mat[r][c++] = ch;
            used[ch - 'A'] = 1;

            if (c == 5) {
                c = 0;
                r++;
            }
        }
    }

    for (ch = 'A'; ch <= 'Z'; ch++) {
        if (!used[ch - 'A']) {
            mat[r][c++] = ch;
            if (c == 5) {
                c = 0;
                r++;
            }
        }
    }
}

void findPos(char ch, int *r, int *c) {
    int i, j;
    if (ch == 'J') ch = 'I';

    for (i = 0; i < 5; i++)
        for (j = 0; j < 5; j++)
            if (mat[i][j] == ch) {
                *r = i;
                *c = j;
                return;
            }
}

int main() {
    char key[100], text[300], p[600], out[600];
    int i, n = 0, len = 0, r1, c1, r2, c2;

    printf("Enter keyword: ");
    scanf("%99s", key);
    getchar();

    printf("Enter plaintext: ");
    fgets(text, sizeof(text), stdin);

    createMatrix(key);

    for (i = 0; text[i]; i++) {
        char ch = toupper((unsigned char)text[i]);
        if (ch >= 'A' && ch <= 'Z') {
            if (ch == 'J') ch = 'I';
            p[n++] = ch;
        }
    }

    for (i = 0; i < n; ) {
        char a = p[i++];
        char b;

        if (i == n) {
            b = (a == 'X') ? 'Q' : 'X';
        } else if (p[i] == a) {
            b = (a == 'X') ? 'Q' : 'X';
        } else {
            b = p[i++];
        }

        findPos(a, &r1, &c1);
        findPos(b, &r2, &c2);

        if (r1 == r2) {
            out[len++] = mat[r1][(c1 + 1) % 5];
            out[len++] = mat[r2][(c2 + 1) % 5];
        } else if (c1 == c2) {
            out[len++] = mat[(r1 + 1) % 5][c1];
            out[len++] = mat[(r2 + 1) % 5][c2];
        } else {
            out[len++] = mat[r1][c2];
            out[len++] = mat[r2][c1];
        }
    }

    out[len] = '\0';

    printf("\nPlayfair Matrix:\n");
    for (i = 0; i < 5; i++) {
        for (int j = 0; j < 5; j++)
            printf("%c ", mat[i][j]);
        printf("\n");
    }

    printf("Ciphertext: %s\n", out);
    return 0;
}
